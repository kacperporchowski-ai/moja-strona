import express from "express";
import cors from "cors";
import OpenAI from "openai";

const app = express();
app.use(cors());
app.use(express.json());

const client = new OpenAI({
    apiKey: process.env.OPENAI_API_KEY
});

// PROSTE OGRANICZENIE (MVP)
let requests = {};

app.post("/generate", async (req, res) => {
    const ip = req.ip;

    // limit free (1 użycie)
    if (requests[ip]) {
        return res.json({ cv: "Limit darmowych CV wykorzystany. Wersja PRO dostępna wkrótce." });
    }

    requests[ip] = true;

    const { name, job, exp } = req.body;

    const prompt = `
Stwórz profesjonalne CV po polsku.

Dane:
Imię: ${name}
Stanowisko: ${job}
Doświadczenie: ${exp}

Format:
- profesjonalny opis
- umiejętności
- doświadczenie
- styl formalny
`;

    const response = await client.chat.completions.create({
        model: "gpt-4o-mini",
        messages: [{ role: "user", content: prompt }]
    });

    res.json({ cv: response.choices[0].message.content });
});

app.listen(3000, () => console.log("CV AI działa"));
